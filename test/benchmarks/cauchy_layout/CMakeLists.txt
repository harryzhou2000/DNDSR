cmake_minimum_required(VERSION 3.21)
project(CauchyLayoutProbe LANGUAGES CXX)
find_package(MPI REQUIRED COMPONENTS CXX)
add_executable(cauchy_layout benchmark.cpp)
target_compile_features(cauchy_layout PRIVATE cxx_std_17)
target_compile_options(cauchy_layout PRIVATE -O3)
target_link_libraries(cauchy_layout PRIVATE MPI::MPI_CXX)
option(CAUCHY_LAYOUT_CUDA "Build the simulated CUDA probe" OFF)
if(CAUCHY_LAYOUT_CUDA)
    enable_language(CUDA)
    add_executable(cauchy_layout_cuda benchmark.cu)
    set_target_properties(cauchy_layout_cuda PROPERTIES CUDA_STANDARD 17)
    target_compile_options(cauchy_layout_cuda PRIVATE -O3)
    target_link_libraries(cauchy_layout_cuda PRIVATE MPI::MPI_CXX)
endif()
